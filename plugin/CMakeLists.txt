# ---------------------------------------------------------------------------
# Qt6 — find before FetchContent so FindBinaryNinjaUI picks up the cached result.

# <repo-root>/qt/ if a previous build-qt run installed Qt there.

# ---------------------------------------------------------------------------
# Binary Ninja API — auto-fetch the exact commit matching the BN install
# ---------------------------------------------------------------------------
include(FetchContent)

# The API commit can be forced via -DBN_API_COMMIT_OVERRIDE=<sha> (the build
# scripts pass this for their --channel / --bn-api flags).  Without it, we fall
# back to the exact commit recorded by the installed Binary Ninja, so the plugin
# ABI matches whatever BN is installed.
if(BN_API_COMMIT_OVERRIDE)
    set(BN_API_COMMIT "${BN_API_COMMIT_OVERRIDE}")
    message(STATUS "Binary Ninja API commit (override): ${BN_API_COMMIT}")
else()
    # On macOS the app bundle keeps api_REVISION.txt under Contents/Resources/,
    # while BN_INSTALL_DIR points at Contents/MacOS/ (where the dylibs live).
    if(APPLE)
        file(READ "${BN_INSTALL_DIR}/../Resources/api_REVISION.txt" _bn_rev_raw)
    else()
        file(READ "${BN_INSTALL_DIR}/api_REVISION.txt" _bn_rev_raw)
    endif()
    # Split into lines, drop blanks, take the last one (the bare SHA-1).
    string(REGEX MATCHALL "[^\r\n]+" _rev_lines "${_bn_rev_raw}")
    list(FILTER _rev_lines EXCLUDE REGEX "^$")
    list(GET _rev_lines -1 BN_API_COMMIT)
    string(STRIP "${BN_API_COMMIT}" BN_API_COMMIT)
    message(STATUS "Binary Ninja API commit (from install): ${BN_API_COMMIT}")
endif()
if(NOT BN_API_COMMIT MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR
        "Could not determine Binary Ninja API commit.\n"
        "Got: '${BN_API_COMMIT}'\n"
        "Pass -DBN_API_COMMIT_OVERRIDE=<sha> or check ${BN_INSTALL_DIR}/api_REVISION.txt")
endif()

FetchContent_Declare(binaryninja-api
    GIT_REPOSITORY https://github.com/Vector35/binaryninja-api.git
    GIT_TAG        ${BN_API_COMMIT}
    GIT_SHALLOW    TRUE
    GIT_SUBMODULES "vendor/fmt"
    SOURCE_DIR     "${CMAKE_SOURCE_DIR}/../binaryninja-api"
    SUBBUILD_DIR   "${CMAKE_BINARY_DIR}/_deps/binaryninja-api-subbuild"
)
FetchContent_MakeAvailable(binaryninja-api)


# ---------------------------------------------------------------------------
# Plugin target
#
# Two link modes:
#
#  - Normal: a Binary Ninja install was found; link the real binaryninjaui
#    (which transitively provides Qt link dirs and core).
#
#  - Stub (CI, no BN install — the Vector35/sample_plugin_cpp approach): the
#    api's BN_ALLOW_STUBS already fools the linker for binaryninjacore; UI
#    symbols are satisfied per platform instead of linking binaryninjaui:
#      macOS  — '-undefined dynamic_lookup' (resolved inside BN at load time)
#      Linux  — shared libs allow undefined symbols by default
#      Windows— an import-lib stub generated from cmake/binaryninjaui_stub.def
#               (regenerate with plugin/tools/gen-ui-stub-def.ps1 if the
#               plugin starts using new UI symbols and the CI link fails)
#    Qt is linked from a stock Qt of the same major.minor; at runtime the
#    imports resolve to the Qt already loaded in the BN process.
#
# BN_FORCE_STUB_LINK=ON selects stub mode even when an install exists — used
# by gen-ui-stub-def.ps1 and for testing the CI configuration locally.
# ---------------------------------------------------------------------------
option(BN_FORCE_STUB_LINK "Force the no-install stub link mode for the plugin target" OFF)

set(_bn_ui_available FALSE)
if(TARGET binaryninjaui AND NOT BN_FORCE_STUB_LINK)
    get_target_property(_bn_ui_type binaryninjaui TYPE)
    if(NOT _bn_ui_type STREQUAL "UTILITY")
        set(_bn_ui_available TRUE)
    endif()
endif()

if(_bn_ui_available)
    add_library(binja-ghidra SHARED
        src/plugin.cpp
        src/BridgeProcess.cpp
        src/BridgeClient.cpp
        src/GhidraConnection.cpp
        src/GhidraJson.cpp
        src/SyncEngine.cpp
        src/ui/ConnectDialog.cpp
        src/ui/ProjectPanel.cpp
        # Weak stubs for Enterprise/Collaboration API — satisfies the static linker
        # when building against standard (non-Enterprise) Binary Ninja.
        src/enterprise_stubs.cpp
        # Headers with Q_OBJECT must be listed so AUTOMOC finds them.
        include/ui/ConnectDialog.h
        include/ui/ProjectPanel.h
    )

    target_include_directories(binja-ghidra PRIVATE
        include
        "${CMAKE_SOURCE_DIR}/../binaryninja-api/vendor"   # nlohmann/json.hpp
    )

    # Bake the CMake-resolved Ghidra path as a compile-time fallback for local mode.
    # The runtime ghidra.ghidraHome setting or auto-detection overrides this.
    if(GHIDRA_HOME)
        target_compile_definitions(binja-ghidra PRIVATE
            GHIDRA_HOME_DEFAULT="${GHIDRA_HOME}")
    endif()

    target_link_libraries(binja-ghidra PRIVATE
        binaryninjaui          # transitively pulls in binaryninjaapi + BN link dirs
        Qt6::Widgets
        Qt6::Core
    )

    if(WIN32)
        target_link_libraries(binja-ghidra PRIVATE ws2_32)
        set_target_properties(binja-ghidra PROPERTIES PREFIX "")
    elseif(UNIX AND NOT APPLE)
        target_link_libraries(binja-ghidra PRIVATE dl)
    endif()

    if(APPLE)
        # Keep the build-time RPATHs in the installed dylib.
        # BN loads plugins via dlopen from its own process, which already has the BN
        # and Qt libs in its address space, so the RPATHs are effectively unused at
        # load time — but removing them on every "cmake --install" run causes
        # install_name_tool to error when the file is "Up-to-date" and the paths
        # were already stripped on a previous install.  BUILD_WITH_INSTALL_RPATH
        # tells CMake not to strip/rewrite RPATHs during install.
        set_target_properties(binja-ghidra PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE)
    endif()

    target_compile_definitions(binja-ghidra PRIVATE
        NOMINMAX
        WIN32_LEAN_AND_MEAN
        _CRT_SECURE_NO_WARNINGS
    )

    # -----------------------------------------------------------------------
    # Install — uses bn_install_plugin() helper defined by the api
    # The bridge JAR is installed alongside the plugin for local mode.
    # In remote mode it lives on the Ghidra server (see server/start-bridge.sh).
    # -----------------------------------------------------------------------
    bn_install_plugin(binja-ghidra)
elseif(BN_ALLOW_STUBS OR BN_FORCE_STUB_LINK)
    message(STATUS "binja-ghidra: stub link mode — UI symbols resolve inside Binary Ninja at load time")

    add_library(binja-ghidra SHARED
        src/plugin.cpp
        src/BridgeProcess.cpp
        src/BridgeClient.cpp
        src/GhidraConnection.cpp
        src/GhidraJson.cpp
        src/SyncEngine.cpp
        src/ui/ConnectDialog.cpp
        src/ui/ProjectPanel.cpp
        src/enterprise_stubs.cpp
        include/ui/ConnectDialog.h
        include/ui/ProjectPanel.h
    )

    target_include_directories(binja-ghidra PRIVATE
        include
        "${CMAKE_SOURCE_DIR}/../binaryninja-api/vendor"   # nlohmann/json.hpp
        "${CMAKE_SOURCE_DIR}/../binaryninja-api/ui"       # UI headers (normally via binaryninjaui)
    )

    if(GHIDRA_HOME)
        target_compile_definitions(binja-ghidra PRIVATE
            GHIDRA_HOME_DEFAULT="${GHIDRA_HOME}")
    endif()

    target_link_libraries(binja-ghidra PRIVATE
        binaryninjaapi
        Qt6::Widgets
        Qt6::Core
    )

    if(WIN32)
        # MSVC needs an import library for every referenced DLL. Build a stub
        # binaryninjaui.lib from the committed .def (symbol names only); the
        # imports bind by DLL name, so they resolve to the real binaryninjaui
        # loaded in the BN process.
        set(_ui_def "${CMAKE_CURRENT_SOURCE_DIR}/cmake/binaryninjaui_stub.def")
        if(NOT EXISTS "${_ui_def}")
            message(FATAL_ERROR "Missing ${_ui_def} — run plugin/tools/gen-ui-stub-def.ps1 to generate it")
        endif()
        set(_ui_stub_lib "${CMAKE_CURRENT_BINARY_DIR}/binaryninjaui_stub/binaryninjaui.lib")
        add_custom_command(
            OUTPUT "${_ui_stub_lib}"
            COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/binaryninjaui_stub"
            COMMAND lib /nologo /def:"${_ui_def}" /machine:x64 /out:"${_ui_stub_lib}"
            DEPENDS "${_ui_def}"
            COMMENT "Generating stub binaryninjaui import library")
        add_custom_target(binaryninjaui-stub-lib DEPENDS "${_ui_stub_lib}")
        add_dependencies(binja-ghidra binaryninjaui-stub-lib)
        target_link_libraries(binja-ghidra PRIVATE "${_ui_stub_lib}" ws2_32)
        set_target_properties(binja-ghidra PROPERTIES PREFIX "")
    elseif(APPLE)
        # UI (and any other) symbols resolve dynamically inside the BN process.
        target_link_options(binja-ghidra PRIVATE "LINKER:-undefined,dynamic_lookup")
        set_target_properties(binja-ghidra PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE)
    else()
        # Linux shared libraries allow undefined symbols by default.
        target_link_libraries(binja-ghidra PRIVATE dl)
    endif()

    target_compile_definitions(binja-ghidra PRIVATE
        NOMINMAX
        WIN32_LEAN_AND_MEAN
        _CRT_SECURE_NO_WARNINGS
    )
else()
    message(STATUS "binja-ghidra: Binary Ninja UI not available — plugin target skipped (test targets still build)")
endif()

# ---------------------------------------------------------------------------
# Tests (Google Test)
#
# The test executable compiles only the Qt-free plugin sources so it can be
# run without a Binary Ninja installation — just the BN API headers/library
# are needed.  Run with:  ctest --test-dir plugin/build -V
# ---------------------------------------------------------------------------
enable_testing()

FetchContent_Declare(googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG        v1.14.0
    GIT_SHALLOW    TRUE
)
# Prevent GoogleTest from overriding our compiler / linker settings on Windows.
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
# Do NOT let GoogleTest add install() rules — it's a private build-time dependency
# and trying to install it to /usr/local requires admin rights and is never wanted.
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

add_executable(binja-ghidra-tests EXCLUDE_FROM_ALL
    test/GhidraConnectionStateTest.cpp
    test/CheckinPreviewTest.cpp
    test/BridgeClientProtocolTest.cpp
    test/GhidraJsonTest.cpp
    # Qt-free production sources only
    src/BridgeClient.cpp
    src/GhidraConnection.cpp
    src/GhidraJson.cpp
    src/BridgeProcess.cpp
    src/enterprise_stubs.cpp
)

target_include_directories(binja-ghidra-tests PRIVATE
    include
    "${CMAKE_SOURCE_DIR}/../binaryninja-api/vendor"   # nlohmann/json.hpp
)

target_link_libraries(binja-ghidra-tests PRIVATE
    GTest::gtest_main
    binaryninjaapi   # required for GhidraConnection.cpp symbol references
)

if(WIN32)
    target_link_libraries(binja-ghidra-tests PRIVATE ws2_32)
elseif(UNIX AND NOT APPLE)
    target_link_libraries(binja-ghidra-tests PRIVATE dl)
endif()

target_compile_definitions(binja-ghidra-tests PRIVATE
    NOMINMAX
    WIN32_LEAN_AND_MEAN
    _CRT_SECURE_NO_WARNINGS
)

include(GoogleTest)
gtest_discover_tests(binja-ghidra-tests)
